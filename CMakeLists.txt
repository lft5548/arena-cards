  if(WIN32 AND NOT ARENA_PROTOBUF_ROOT AND EXISTS "D:/Dev/protobuf/bin/protoc.exe")
    set(ARENA_PROTOBUF_ROOT "D:/Dev/protobuf")
  endif()
  list(APPEND CMAKE_PREFIX_PATH ${ARENA_PROTOBUF_ROOT})
  if(WIN32)
    set(Protobuf_USE_STATIC_LIBS ON)
  endif()
  find_package(Protobuf 3.21.12 EXACT REQUIRED)
  file(RELATIVE_PATH ARENA_PROTO_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/proto)
  set(ARENA_PROTO_SOURCES ${CMAKE_CURRENT_BINARY_DIR}/arena_cards.pb.cc)
  set(ARENA_PROTO_HEADERS ${CMAKE_CURRENT_BINARY_DIR}/arena_cards.pb.h)
  add_custom_command(OUTPUT ${ARENA_PROTO_SOURCES} ${ARENA_PROTO_HEADERS}
    COMMAND protobuf::protoc --cpp_out=. --proto_path=${ARENA_PROTO_DIRECTORY}
            ${ARENA_PROTO_DIRECTORY}/arena_cards.proto
    DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/proto/arena_cards.proto
    WORKING_DIRECTORY ${CMAKE_CURRENT_BINARY_DIR} VERBATIM)
  add_library(arena_protobuf STATIC ${ARENA_PROTO_SOURCES} server/gateway/protobuf_codec.cpp)
  target_include_directories(arena_protobuf PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_BINARY_DIR})
  target_link_libraries(arena_protobuf PUBLIC protobuf::libprotobuf)
endif()
